rules_version = '2'; service cloud.firestore { match /databases/{database}/documents { // Memastikan pengguna sudah login melalui Firebase Authentication. function signedIn() { return request.auth != null; } // Memeriksa apakah profil pengguna tersedia. function userExists(uid) { return exists( /databases/$(database)/documents/users/$(uid) ); } // Admin dikenali dari peran pada dokumen users/{UID}. function isAdmin() { return signedIn() && userExists(request.auth.uid) && get( /databases/$(database)/documents/users/$(request.auth.uid) ).data.role == "admin"; } // Guru hanya dapat menggunakan fitur guru jika disetujui admin. function isApprovedTeacher() { return signedIn() && userExists(request.auth.uid) && get( /databases/$(database)/documents/users/$(request.auth.uid) ).data.role == "guru" && get( /databases/$(database)/documents/users/$(request.auth.uid) ).data.approved == true; } // ===================================== // KOLEKSI USERS // ===================================== match /users/{uid} { // Pengguna hanya dapat membuat profil untuk UID miliknya. // Pendaftaran umum hanya mengizinkan role guru atau murid. // Role admin harus ditetapkan secara tepercaya melalui Console // atau backend tepercaya, bukan dari formulir pendaftaran. allow create: if signedIn() && request.auth.uid == uid && request.resource.data.role in ["guru", "murid"] && request.resource.data.approved == (request.resource.data.role == "murid") && request.resource.data.email == request.auth.token.email; // Pengguna membaca profilnya sendiri. // Admin dapat membaca profil semua pengguna. allow read: if signedIn() && ( request.auth.uid == uid || isAdmin() ); // Admin hanya boleh mengubah status persetujuan. // Field lain, termasuk role dan email, tidak dapat diubah // melalui operasi update ini. allow update: if isAdmin() && request.resource.data.diff(resource.data) .affectedKeys().hasOnly(["approved"]) && request.resource.data.approved is bool; // Hanya admin yang dapat menghapus profil pengguna. allow delete: if isAdmin(); } // ===================================== // KOLEKSI CLASSES // ===================================== match /classes/{classId} { // Guru yang sudah disetujui dapat membuat kelas // atas nama UID miliknya sendiri. allow create: if isApprovedTeacher() && request.resource.data.teacherId == request.auth.uid; // Admin dapat membaca seluruh kelas. // Guru hanya dapat membaca kelas miliknya. allow read: if isAdmin() || ( isApprovedTeacher() && resource.data.teacherId == request.auth.uid ); // Admin dapat mengubah atau menghapus kelas mana pun. // Guru hanya dapat mengubah atau menghapus kelas miliknya. allow update, delete: if isAdmin() || ( isApprovedTeacher() && resource.data.teacherId == request.auth.uid ); } // ===================================== // KOLEKSI MATERIALS // ===================================== match /materials/{materialId} { // Semua pengguna yang login dapat membaca materi. allow read: if signedIn(); // Hanya admin yang dapat menambah, mengubah, // atau menghapus materi. allow create, update, delete: if isAdmin(); } // ===================================== // TOLAK AKSES KE KOLEKSI LAIN // ===================================== match /{document=**} { allow read, write: if false; } } }
